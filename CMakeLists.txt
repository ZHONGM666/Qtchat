cmake_minimum_required(VERSION 3.16)
project(QtChat VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED)

option(QTCHAT_BUILD_CLIENT "Build the Qt Widgets client" ON)
option(QTCHAT_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(QTCHAT_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(QTCHAT_BUILD_CLIENT)
    find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
endif()

add_library(qtchat_common common/frame.cpp)
target_include_directories(qtchat_common PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(qtchat_common PRIVATE /W4 /utf-8)
else()
    target_compile_options(qtchat_common PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(qtchat_runtime common/idle_timeout.cpp)
target_include_directories(qtchat_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/thread-pool/include")
target_link_libraries(qtchat_runtime PUBLIC Threads::Threads Boost::headers)
if(QTCHAT_BUILD_CLIENT)
    add_library(qtchat_network client/networkclient.h client/networkclient.cpp)
    set_target_properties(qtchat_network PROPERTIES AUTOMOC ON)
    target_link_libraries(qtchat_network PUBLIC qtchat_common qtchat_runtime Qt6::Network)
    add_library(qtchat_ui mainwindow.h mainwindow.cpp client/loginpage.h client/loginpage.cpp client/chatpage.h client/chatpage.cpp client/icons.h client/resources.qrc)
    set_target_properties(qtchat_ui PROPERTIES AUTOMOC ON AUTORCC ON)
    target_link_libraries(qtchat_ui PUBLIC qtchat_network Qt6::Widgets)
    add_executable(qtchat_client main.cpp)
    target_link_libraries(qtchat_client PRIVATE qtchat_ui)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    find_package(SQLite3 REQUIRED)
    add_library(qtchat_store server/password.cpp server/database.cpp server/service.cpp)
    target_include_directories(qtchat_store PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/json/single_include")
    target_link_libraries(qtchat_store PUBLIC qtchat_runtime OpenSSL::Crypto SQLite::SQLite3)
    add_executable(qtchat_server server/main.cpp server/reactor.cpp)
    target_link_libraries(qtchat_server PRIVATE qtchat_store qtchat_common)
endif()

if(BUILD_TESTING)
    if(QTCHAT_BUILD_CLIENT)
        find_package(Qt6 REQUIRED COMPONENTS Test)
        add_executable(networkclient_test tests/networkclient_test.cpp)
        set_target_properties(networkclient_test PROPERTIES AUTOMOC ON)
        target_link_libraries(networkclient_test PRIVATE qtchat_network Qt6::Test)
        add_test(NAME networkclient_test COMMAND networkclient_test)
        add_test(NAME network_invalid_response COMMAND networkclient_test invalidResponseStillCompletesTheAffectedRequest)
        add_executable(chat_state_test tests/chat_state_test.cpp)
        set_target_properties(chat_state_test PROPERTIES AUTOMOC ON)
        target_link_libraries(chat_state_test PRIVATE qtchat_ui Qt6::Test)
        add_test(NAME chat_state_test COMMAND chat_state_test)
        set_tests_properties(chat_state_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        set_tests_properties(networkclient_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        if(TARGET qtchat_server)
            add_executable(client_ui_test tests/client_ui_test.cpp)
            set_target_properties(client_ui_test PROPERTIES AUTOMOC ON)
            target_link_libraries(client_ui_test PRIVATE qtchat_ui Qt6::Test)
            add_test(NAME client_ui_test COMMAND client_ui_test)
            set_tests_properties(client_ui_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QTCHAT_SERVER=$<TARGET_FILE:qtchat_server>;QTCHAT_OUTPUT_DIR=${CMAKE_CURRENT_SOURCE_DIR}/tests/results/ui" TIMEOUT 90)
            add_test(NAME client_visual_smoke COMMAND client_ui_test liveTwoClients)
            set_tests_properties(client_visual_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=xcb;QTCHAT_SERVER=$<TARGET_FILE:qtchat_server>;QTCHAT_OUTPUT_DIR=${CMAKE_CURRENT_SOURCE_DIR}/tests/results/ui-native" TIMEOUT 90 LABELS "gui")
        endif()
    endif()
    add_executable(frame_test tests/frame_test.cpp)
    target_link_libraries(frame_test PRIVATE qtchat_common)
    add_test(NAME frame_test COMMAND frame_test)
    add_executable(runtime_test tests/runtime_test.cpp)
    target_link_libraries(runtime_test PRIVATE qtchat_runtime)
    add_test(NAME runtime_test COMMAND runtime_test)
    if(TARGET qtchat_store)
        add_executable(database_test tests/database_test.cpp)
        target_link_libraries(database_test PRIVATE qtchat_store)
        add_test(NAME database_test COMMAND database_test)
        add_executable(service_test tests/service_test.cpp)
        target_link_libraries(service_test PRIVATE qtchat_store)
        add_test(NAME service_test COMMAND service_test)
    endif()
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            add_test(NAME harness_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/harness_test.py")
        endif()
        add_test(NAME linux_helper_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/linux_helper_test.py")
        add_test(NAME stress_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/stress_test.py")
        set_tests_properties(stress_test PROPERTIES TIMEOUT 20)
        if(TARGET qtchat_server)
            add_test(NAME stress_smoke COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/stress_smoke.py" --server $<TARGET_FILE:qtchat_server>)
            set_tests_properties(stress_smoke PROPERTIES TIMEOUT 45)
            add_test(NAME end_to_end COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/end_to_end.py" --server $<TARGET_FILE:qtchat_server>)
            set_tests_properties(end_to_end PROPERTIES TIMEOUT 600)
            add_test(NAME benchmark_suite COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_benchmarks.py" --server $<TARGET_FILE:qtchat_server> --release)
            set_tests_properties(benchmark_suite PROPERTIES TIMEOUT 800 SKIP_RETURN_CODE 77 LABELS "performance")
            add_test(NAME reactor_integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/reactor_integration.py" --server $<TARGET_FILE:qtchat_server>)
            set_tests_properties(reactor_integration PROPERTIES TIMEOUT 60)
        endif()
    endif()
endif()
